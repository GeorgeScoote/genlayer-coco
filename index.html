<script>

    document.addEventListener('DOMContentLoaded', () => {

        const playerCountSpan = document.getElementById('player-count');

        const playerListUl = document.getElementById('player-list');

        const playerListContainer = document.getElementById('player-list-container');

        const wordInput = document.getElementById('word-input');

        const submitWordBtn = document.getElementById('submit-word-btn');

        const refreshStatusBtn = document.getElementById('refresh-status-btn');

        const leaderboardListUl = document.getElementById('leaderboard-list').querySelector('ul');

        const refreshLeaderboardBtn = document.getElementById('refresh-leaderboard-btn');


        async function updateGameStatusUI() {

            playerListContainer.innerHTML = '<p class="loader">正在刷新游戏状态...</p>';

            submitWordBtn.disabled = true;

            const status = await getGameStatus();

            playerListUl.innerHTML = '';

            playerListContainer.innerHTML = '';

            if (status) {

                playerCountSpan.textContent = status.players;

                if (status.players > 0) {

                    status.addresses.forEach((address, index) => {

                        const word = status.words[index];

                        const li = document.createElement('li');

                        li.innerHTML = `<div>玩家: <span class="player-address">${address.substring(0, 10)}...</span></div> <div>提交的单词: <span class="player-word">${word}</span></div>`;

                        playerListUl.appendChild(li);

                    });

                } else { playerListContainer.innerHTML = '<p>当前没有玩家，快来加入吧！</p>'; }

            } else { playerListContainer.innerHTML = '<p>获取游戏状态失败，请检查钱包连接或刷新页面。</p>'; }

            submitWordBtn.disabled = false;

        }


        async function updateLeaderboardUI() {

            leaderboardListUl.innerHTML = '<p class="loader">正在刷新排行榜...</p>';

            const board = await getLeaderboard();

            leaderboardListUl.innerHTML = '';

            if (board && Object.keys(board).length > 0) {

                const sortedBoard = Object.entries(board).sort(([, a], [, b]) => b - a);

                sortedBoard.forEach(([address, score]) => {

                     const li = document.createElement('li');

                     li.innerHTML = `<span>${address.substring(0, 10)}...</span> <strong>${score} 分</strong>`;

                     leaderboardListUl.appendChild(li);

                });

            } else { leaderboardListUl.innerHTML = '<p>排行榜上还没有数据。</p>'; }

        }


        submitWordBtn.addEventListener('click', async () => {

            const word = wordInput.value;

            if (!word) { alert('请输入一个单词！'); return; }

            submitWordBtn.disabled = true;

            submitWordBtn.textContent = '提交中...';

            const result = await submitWord(word);

            if (result) { setTimeout(() => { updateGameStatusUI(); updateLeaderboardUI(); wordInput.value = ''; }, 3000); }

            submitWordBtn.disabled = false;

            submitWordBtn.textContent = '提交单词';

        });


        refreshStatusBtn.addEventListener('click', updateGameStatusUI);

        refreshLeaderboardBtn.addEventListener('click', updateLeaderboardUI);


        updateGameStatusUI();

        updateLeaderboardUI();

        setInterval(updateGameStatusUI, 15000);

    });

</script>
